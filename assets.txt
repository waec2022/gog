##### FILE: src/js/app.js #####
/*
 * Campus Guide public app: live search, filters, detail pages, lazy images, documents, device cache.
 * Vanilla JS, no dependencies, no third-party requests. Record types and their labels come from the data
 * (manifest.types), so Custom Features created in the Editor appear here without code changes.
 *
 * Device cache (the phone is a CACHE, never the master database):
 *   - Service Worker (sw.js): app shell, versioned data files, viewed images.
 *   - IndexedDB: last manifest, the search indexes that were actually used, the home page, recently opened records.
 *   - Every cached item is tied to the data version (manifest.v). A new version is detected on open, on return to the
 *     tab and every 5 minutes; changed indexes and the open page are refreshed in place.
 */
(function () {
  'use strict';
  var S = window.CGSearch;
  var PAGE = 30, RECENT_MAX = 40;
  var LABEL = { office: 'Office', lecturer: 'Lecturer', course: 'Course', department: 'Department', faculty: 'Faculty', service: 'Service', building: 'Building', location: 'Location', admission: 'Admissions', fee: 'School Fees', timetable: 'Timetable', calendar: 'Academic Calendar', announcement: 'Announcement', event: 'Event', campaign: 'Campaign', notice: 'Important Notice', live: 'Live Update', document: 'Document' };
  var VER = { verified: 'Verified Information', verified_staff: 'Verified Staff', official_staff: 'Official Staff', pending: 'Pending Verification', unverified: 'Unverified' };
  var FILTERS = ['all', 'offices', 'lecturers', 'courses', 'departments', 'faculties', 'services', 'buildings', 'information'];
  var BUILTIN_HOME = ['events', 'campaigns', 'announcements', 'updates'];

  var $ = function (id) { return document.getElementById(id); };
  var M = null, manifestP = null, indexes = {}, pending = {}, indexVer = {};
  var cur = { q: '', f: 'all' }, shown = PAGE, last = [], liveList = [], urlTimer = 0, record = null, curPath = '';

  function esc(s) {
    return String(s == null ? '' : s).replace(/[&<>"']/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]; });
  }
  function fetchJSON(url, opt) {
    return fetch(url, opt).then(function (r) {
      if (!r.ok) { var e = new Error('HTTP ' + r.status); e.status = r.status; throw e; }
      return r.json();
    });
  }
  function labelOf(t) { return (M && M.types && M.types[t]) || LABEL[t] || t; }

  /* ---------- IndexedDB key/value (best effort: the site works if it is unavailable) ---------- */
  var dbP = null;
  function idb() {
    if (!dbP) dbP = new Promise(function (res) {
      try {
        var rq = indexedDB.open('campus-guide', 1);
        rq.onupgradeneeded = function () { rq.result.createObjectStore('kv'); };
        rq.onsuccess = function () { res(rq.result); };
        rq.onerror = function () { res(null); };
      } catch (e) { res(null); }
    });
    return dbP;
  }
  function kvGet(k) {
    return idb().then(function (d) {
      if (!d) return null;
      return new Promise(function (res) {
        try { var r = d.transaction('kv').objectStore('kv').get(k); r.onsuccess = function () { res(r.result || null); }; r.onerror = function () { res(null); }; } catch (e) { res(null); }
      });
    });
  }
  function kvSet(k, v) {
    return idb().then(function (d) { if (d) try { d.transaction('kv', 'readwrite').objectStore('kv').put(v, k); } catch (e) { /* quota: ignore */ } });
  }
  function kvDel(k) {
    return idb().then(function (d) { if (d) try { d.transaction('kv', 'readwrite').objectStore('kv').delete(k); } catch (e) { /* ignore */ } });
  }

  /* ---------- images and files: ONE place to change where media comes from ---------- */
  // ref is either an uploaded media key ("img/202610/ab12.webp") or a legacy external URL.
  function mediaUrl(ref, size) {
    if (!ref) return '';
    if (/^https?:\/\//i.test(ref)) return ref;
    var base = (M && M.media && M.media.base) || '';
    var k = size === 't' ? ref.replace(/\.(webp|jpe?g|png|gif)$/i, '.t.$1') : ref;
    return base + '/' + k;
  }
  // size 't' = small thumbnail (lists, avatars); anything else = full-size
  function imgTag(ref, alt, size) {
    var full = mediaUrl(ref, 'f');
    return '<img src="' + esc(mediaUrl(ref, size)) + '" data-full="' + (size === 't' ? esc(full) : '') + '" alt="' + esc(alt || '') + '" loading="lazy" decoding="async" referrerpolicy="no-referrer">';
  }
  // A missing thumbnail falls back to the full image once; a broken image never breaks the page.
  document.addEventListener('error', function (e) {
    var t = e.target;
    if (!t || t.tagName !== 'IMG' || !t.parentNode) return;
    var full = t.getAttribute('data-full');
    if (full && !t.getAttribute('data-tried')) { t.setAttribute('data-tried', '1'); t.src = full; return; }
    t.parentNode.className += ' broken';
  }, true);

  /* ---------- data ---------- */
  function applyManifest(m) { M = m; return m; }
  function manifest() {
    if (!manifestP) {
      manifestP = fetchJSON('/data/index.json', { cache: 'no-cache' }).then(function (m) { kvSet('manifest', m); return applyManifest(m); }, function (err) {
        return kvGet('manifest').then(function (m) { if (m) return applyManifest(m); throw err; });
      });
    }
    return manifestP;
  }
  function vq(u) { return '/' + u + '?v=' + M.v; }
  function groupTypes(id) {
    for (var i = 0; i < M.groups.length; i++) if (M.groups[i].id === id) return M.groups[i].types;
    return [];
  }
  function fetchRaw(urls) {
    return Promise.all(urls.map(function (u) { return fetchJSON(vq(u)); })).then(function (parts) { return [].concat.apply([], parts); });
  }
  function urlsFor(f) {
    if (f === 'all') return M.all || [].concat.apply([], Object.keys(M.chunks).map(function (k) { return M.chunks[k]; }));
    return M.chunks[f] || [];
  }
  function fetchIndex(f) {
    return manifest().then(function () { return fetchRaw(urlsFor(f)); }).then(function (raw) {
      kvSet('idx:' + f, { v: M.v, e: raw }); // store the raw entries before prepare() adds search fields
      indexVer[f] = M.v;
      return S.prepare(raw);
    });
  }
  // Returns instantly from the device cache when this index was used before, then checks the version in the background.
  function loadIndex(f) {
    if (indexes[f]) return Promise.resolve(indexes[f]);
    if (pending[f]) return pending[f];
    var p;
    if (f !== 'all' && indexes.all) {
      var ts = groupTypes(f);
      p = Promise.resolve(indexes.all.filter(function (e) { return ts.indexOf(e.t) >= 0; }));
    } else {
      p = kvGet('idx:' + f).then(function (c) {
        if (c && c.e) {
          indexVer[f] = c.v;
          manifest().then(function () { if (c.v !== M.v) refreshIndex(f); }, function () { /* offline: keep cached */ });
          return S.prepare(c.e);
        }
        return fetchIndex(f);
      });
    }
    pending[f] = p.then(function (list) { indexes[f] = list; delete pending[f]; return list; }, function (e) { delete pending[f]; throw e; });
    return pending[f];
  }
  function refreshIndex(f) {
    return fetchIndex(f).then(function (list) {
      indexes[f] = list;
      if (cur.f === f || f === 'all') run(true);
    }, function () { /* keep the cached copy */ });
  }

  /* ---------- helpers ---------- */
  function slugOf(id, type) { return id.slice(type.length + 1); }
  function pathOf(type, id) { return '/' + type + '/' + slugOf(id, type); }
  function fmtDate(d) {
    var m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(d || '');
    if (!m) return '';
    return new Date(+m[1], +m[2] - 1, +m[3]).toLocaleDateString(undefined, { day: 'numeric', month: 'short', year: 'numeric' });
  }
  function left(e) {
    var s = Math.max(0, Math.round((e * 1000 - Date.now()) / 1000));
    if (s < 3600) return Math.max(1, Math.ceil(s / 60)) + ' min left';
    if (s < 86400) return Math.floor(s / 3600) + 'h ' + Math.floor((s % 3600) / 60) + 'm left';
    return Math.floor(s / 86400) + ' days left';
  }
  function paras(t) {
    return String(t || '').split(/\n+/).filter(Boolean).map(function (p) { return '<p>' + esc(p) + '</p>'; }).join('');
  }
  function fmtSize(n) { return !n ? '' : n < 1024 ? n + ' B' : n < 1048576 ? Math.round(n / 1024) + ' KB' : (n / 1048576).toFixed(1) + ' MB'; }

  /* ---------- search view ---------- */
  function resultHTML(e) {
    var v = e.v ? '<u>' + esc(VER[e.v] || 'Verified') + '</u>' : '';
    return '<li><a class="r" href="' + pathOf(e.t, e.i) + '"><div class="l1"><b>' + esc(e.n) + '</b><i>' + esc(labelOf(e.t)) + '</i></div>' +
      (v || e.s ? '<div class="l2">' + v + esc(e.s || '') + '</div>' : '') + '</a></li>';
  }
  function paint(msg) {
    var n = last.length, q = cur.q.trim();
    $('status').textContent = msg || (n ? n + (n === 1 ? ' result' : ' results') : 'No results for “' + q + '”. Try different words or choose All.');
    var html = '';
    for (var i = 0; i < Math.min(n, shown); i++) html += resultHTML(last[i]);
    $('results').innerHTML = html;
    $('more').hidden = n <= shown;
  }
  function run(keepShown) {
    var q = cur.q, f = cur.f, has = !!q.trim();
    $('search-view').className = has ? 'searching' : '';
    $('clear').hidden = !q;
    $('home').hidden = has || f !== 'all';
    if (!has && f === 'all') { last = []; $('results').innerHTML = ''; $('more').hidden = true; $('status').textContent = ''; return; }
    if (!indexes[f] && !(f !== 'all' && indexes.all)) $('status').textContent = 'Loading search…';
    loadIndex(f).then(function (list) {
      if (cur.q !== q || cur.f !== f) return; // user kept typing; a newer run is already in charge
      var types = groupTypes(f);
      var keep = keepShown ? shown : PAGE;
      if (has) {
        last = S.search(list, q, { types: types, now: Date.now() });
        shown = keep;
        paint();
      } else {
        last = S.browse(list, types, Date.now());
        shown = keep;
        paint(last.length ? 'Showing ' + last.length + ' · type to search within this category' : 'Nothing published here yet.');
      }
    }).catch(function () {
      if (cur.q === q && cur.f === f) $('status').textContent = 'Search could not load. Check your connection and try again.';
    });
  }
  function searchUrl() {

/*
 * Campus Guide search engine — no AI, just normalisation + weighted matching.
 * Index entry keys: i id, t type, n title, s subtitle, k keywords ('|' separated),
 * a aliases ('|' separated), x related-record text, d short description, e expiry (epoch seconds, 0 = never),
 * v verification status (only when verified).
 */
(function (root, factory) {
  if (typeof module === 'object' && module.exports) module.exports = factory();
  else root.CGSearch = factory();
})(typeof self !== 'undefined' ? self : this, function () {
  'use strict';

  // Words that add little meaning in natural questions. Never removed if nothing else would be left.
  var FILLER = {};
  ('where is the who teaches teach taught what are can i find please of for to me show do does in at which belong belongs own owns located').split(' ')
    .forEach(function (w) { FILLER[w] = 1; });

  // Words that name a record type: used to boost that type rather than to match text.
  // [type, isPlural]. A plural ("which courses...") means the person wants a list of that type, so it outranks a
  // title match of another type; a singular ("computer science department") only nudges that type up.
  var TYPEWORD = {
    office: ['office', 0], offices: ['office', 1], lecturer: ['lecturer', 0], lecturers: ['lecturer', 1], course: ['course', 0], courses: ['course', 1],
    department: ['department', 0], departments: ['department', 1], dept: ['department', 0], faculty: ['faculty', 0], faculties: ['faculty', 1],
    service: ['service', 0], services: ['service', 1], building: ['building', 0], buildings: ['building', 1],
    timetable: ['timetable', 0], timetables: ['timetable', 1], announcement: ['announcement', 0], announcements: ['announcement', 1],
    event: ['event', 0], events: ['event', 1], campaign: ['campaign', 0], campaigns: ['campaign', 1], notice: ['notice', 0], notices: ['notice', 1]
  };

  function norm(s) {
    return String(s || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '')
      .replace(/[^a-z0-9]+/g, ' ').trim();
  }
  function words(s) { return s ? s.split(' ') : []; }
  function list(s) {
    if (!s) return [];
    return s.split('|').map(norm).filter(Boolean);
  }
  function flat(arr) {
    var o = [];
    arr.forEach(function (p) { words(p).forEach(function (w) { o.push(w); }); });
    return o;
  }

  // Precompute normalised fields once after the index is downloaded.
  function prepare(entries) {
    for (var i = 0; i < entries.length; i++) {
      var e = entries[i];
      if (e._n !== undefined) continue;
      e._n = norm(e.n);
      e._nw = words(e._n);
      e._nc = e._n.replace(/ /g, '');
      e._al = list(e.a);
      e._aw = flat(e._al);
      e._kl = list(e.k);
      e._kw = flat(e._kl);
      e._sw = words(norm(e.s));
      e._xw = words(norm(e.x));
      e._dw = words(norm(e.d));
    }
    return entries;
  }

  function parse(raw) {
    var all = words(norm(raw));
    var tokens = all.filter(function (t) { return !FILLER[t]; });
    if (!tokens.length) tokens = all; // all fillers: keep them rather than search nothing
    var boost = {};
    var rest = tokens.filter(function (t) {
      if (TYPEWORD[t]) { var tw = TYPEWORD[t]; boost[tw[0]] = Math.max(boost[tw[0]] || 0, tw[1] ? 1100 : 200); return false; }
      return true;
    });
    if (rest.length) tokens = rest; else { if (Object.keys(boost).length) tokens = []; }
    var lower = String(raw || '').toLowerCase();
    if (/\bwho\b.*\b(teach|lectur)|\btaught by\b/.test(lower)) boost.lecturer = Math.max(boost.lecturer || 0, 1200);
    return { tokens: tokens, phrase: tokens.join(' '), boost: boost, where: /^\s*where\b/.test(lower) };
  }

  function best(t, wordsArr, exact, prefix) {
    var b = 0;
    for (var i = 0; i < wordsArr.length; i++) {
      var w = wordsArr[i];
      if (w === t) return exact;
      if (b < prefix && w.indexOf(t) === 0) b = prefix;
    }
    return b;
  }

  function scoreEntry(e, q) {
    var tokens = q.tokens, n = tokens.length, matched = 0, score = 0, i, j;
    var need = n <= 3 ? n : Math.ceil(n * 0.6);
    for (i = 0; i < n; i++) {
      var t = tokens[i];
      var b = best(t, e._nw, 100, 80);
      if (b < 30 && t.length >= 3 && e._n.indexOf(t) >= 0) b = 30;
      if (b < 100) b = Math.max(b, best(t, e._kw, 60, 50));
      if (b < 100) b = Math.max(b, best(t, e._aw, 70, 60));
      if (b < 60) b = Math.max(b, best(t, e._sw, 35, 30));
      if (b < 30) b = Math.max(b, best(t, e._xw, 18, 14));
      if (b < 14) b = Math.max(b, best(t, e._dw, 6, 5));
      if (b > 0) { matched++; score += b; }
    }
    if (matched < need) return 0;
    var p = q.phrase, bonus = 0;
    if (e._n === p) bonus = 1000;
    else if (e._n.indexOf(p) === 0) bonus = 800;
    else if (p.length >= 2 && e._nc.indexOf(p.replace(/ /g, '')) === 0) bonus = 700;
    else if ((' ' + e._n).indexOf(' ' + p) >= 0) bonus = 500;
    for (j = 0; j < e._al.length && bonus < 750; j++) {
      if (e._al[j] === p) bonus = Math.max(bonus, 750);
      else if (e._al[j].indexOf(p) === 0) bonus = Math.max(bonus, 550);
    }
    if (bonus < 400) for (j = 0; j < e._kl.length; j++) { if (e._kl[j] === p) { bonus = 400; break; } }
    score += bonus;
    if (e.v) score += 3;
    return score;
  }

  // entries: prepared index entries. opts: { types: [..]|null, now: ms, limit }
  function search(entries, raw, opts) {
    opts = opts || {};
    var q = parse(raw);
    var now = opts.now || Date.now();
    var allowed = null;
    if (opts.types) { allowed = {}; opts.types.forEach(function (t) { allowed[t] = 1; }); }
    var out = [];
    var boost = q.boost, hasBoost = Object.keys(boost).length > 0;
    for (var i = 0; i < entries.length; i++) {
      var e = entries[i];
      if (allowed && !allowed[e.t]) continue;
      if (e.e && e.e * 1000 <= now) continue;
      if (e.b && e.b * 1000 > now) continue; // live update that has not started yet
      var s;
      if (!q.tokens.length) s = boost[e.t] ? 100 : 0; // only a type word was typed, e.g. "offices"
      else s = scoreEntry(e, q);
      if (!s) continue;
      if (boost[e.t]) s += boost[e.t];
      if (q.where && (e.t === 'office' || e.t === 'building' || e.t === 'location' || e.t === 'lecturer')) s += 20;
      out.push({ s: s, e: e });
    }
    out.sort(function (a, b) { return b.s - a.s || (a.e.n < b.e.n ? -1 : a.e.n > b.e.n ? 1 : 0); });
    var res = out.map(function (o) { return o.e; });
    return opts.limit ? res.slice(0, opts.limit) : res;
  }

  // Alphabetical listing of a category when nothing has been typed.
  function browse(entries, types, now) {
    var allowed = {};
    (types || []).forEach(function (t) { allowed[t] = 1; });
    now = now || Date.now();
    return entries.filter(function (e) { return allowed[e.t] && !(e.e && e.e * 1000 <= now) && !(e.b && e.b * 1000 > now); })
      .sort(function (a, b) { return a._n < b._n ? -1 : a._n > b._n ? 1 : 0; });
  }

  return { norm: norm, prepare: prepare, search: search, browse: browse, parse: parse };
});

/*
 * Campus Guide service worker.
 *  - App shell + build-versioned data files: cached per build (cache name changes with every build).
 *  - Unversioned data (manifest, home, live): network first, cache only as an offline fallback.
 *  - Viewed images: cached in a small separate cache (max MEDIA_MAX) so opened images load instantly next time.
 *  - Documents (/doc/) are never cached automatically.
 * Nothing here can hold new campus information back: manifest.v changes on every publish and all versioned URLs include it.
 */
var V = 'cg-__BUILD__';
var MEDIA = 'cg-media';
var MEDIA_MAX = 150;
var SHELL = ['/', '/css/app.css?v=__BUILD__', '/js/search.js?v=__BUILD__', '/js/app.js?v=__BUILD__', '/favicon.svg'];
var IMG_RE = /\/img\/[A-Za-z0-9_\/.-]+\.(webp|jpe?g|png|gif)$/;

self.addEventListener('install', function (e) {
  e.waitUntil(caches.open(V).then(function (c) { return c.addAll(SHELL); }).then(function () { return self.skipWaiting(); }));
});
self.addEventListener('activate', function (e) {
  e.waitUntil(caches.keys().then(function (keys) {
    return Promise.all(keys.filter(function (k) { return k.indexOf('cg-') === 0 && k !== V && k !== MEDIA; }).map(function (k) { return caches.delete(k); }));
  }).then(function () { return self.clients.claim(); }));
});

function net(req) {
  return new Promise(function (resolve, reject) {
    var t = setTimeout(reject, 5000);
    fetch(req).then(function (r) { clearTimeout(t); resolve(r); }, function (err) { clearTimeout(t); reject(err); });
  });
}
function trim(cache) {
  return cache.keys().then(function (keys) {
    if (keys.length <= MEDIA_MAX) return;
    return Promise.all(keys.slice(0, keys.length - MEDIA_MAX).map(function (k) { return cache.delete(k); }));
  });
}
// Image request from the page (no-cors). Retry as CORS so the response is a normal, cacheable one.
function image(req) {
  return caches.open(MEDIA).then(function (c) {
    return c.match(req.url).then(function (hit) {
      if (hit) return hit;
      return fetch(req.url, { mode: 'cors', credentials: 'omit' }).then(function (r) {
        if (r.ok) { c.put(req.url, r.clone()).then(function () { return trim(c); }); }
        return r;
      }).catch(function () { return fetch(req); });
    });
  });
}

self.addEventListener('fetch', function (e) {
  var req = e.request;
  if (req.method !== 'GET') return;
  var url = new URL(req.url);
  if (req.destination === 'image' && IMG_RE.test(url.pathname)) { e.respondWith(image(req)); return; }
  if (url.origin !== location.origin) return;
  var versioned = url.pathname.indexOf('/data/') === 0 && url.search.indexOf('v=') >= 0 && url.pathname.indexOf('/data/live') !== 0;
  if (versioned) { // content-addressed by build id: cache first
    e.respondWith(caches.open(V).then(function (c) {
      return c.match(req).then(function (hit) {
        return hit || fetch(req).then(function (r) { if (r.ok) c.put(req, r.clone()); return r; });
      });
    }));
    return;
  }
  e.respondWith(net(req).then(function (r) {
    if (r.ok) { var copy = r.clone(); caches.open(V).then(function (c) { c.put(req, copy); }); }
    return r;
  }).catch(function () {
    return caches.match(req).then(function (hit) {
      return hit || (req.mode === 'navigate' ? caches.match('/') : Response.error());
    });
  }));
});

      if (!r.ok) throw new Error('The Worker answered with an error (' + r.status + ').');
      return r.json();
    }).then((j) => {
      localStorage.setItem('cg-api', st.api); sessionStorage.setItem('cg-key', st.key);
      loadFiles(j.files || {}); st.busy = false; st.connectMsg = ''; render();
    }).catch((e) => { st.busy = false; st.connectMsg = e.message === 'Failed to fetch' ? 'Could not reach the Worker. Check the URL and your connection.' : e.message; render(); });
  }
  function loadBundle(file) {
    const rd = new FileReader();
    rd.onload = () => {
      try {
        const j = JSON.parse(rd.result);
        const files = j && j.format === 'campus-guide-source' ? j.files : j;
        if (!files || typeof files !== 'object' || !Object.keys(files).some((k) => k.indexOf('data/source/') === 0)) throw new Error('Not a Campus Guide data.json file');
        loadFiles(files); render();
      } catch (e) { st.connectMsg = 'That file could not be read: ' + e.message; render(); }
    };
    rd.readAsText(file);
  }

  function usedKeys() {
    const set = new Set();
    const walk = (o) => {
      if (typeof o === 'string') { if (S.isImageKey(o) || S.isDocKey(o)) set.add(o); }
      else if (Array.isArray(o)) o.forEach(walk);

    };
    walk(currentFiles());
    return set;
  }

  function closeOverlay() { overlay.hidden = true; overlay.innerHTML = ''; }
  function uniqueId(type, title) {
    const base = type + '-' + (S.slugify(title) || 'new');
    let id = base, n = 2;
    while (find(id)) id = base + '-' + n++;
    return id;
  }

  function previewSheet(rec, T) {
    const el = document.createElement('div');

      else if (fd.t === 'files') v.forEach((f) => files.push(f));
      else if (fd.t === 'textarea') blocks.push([fd.l, v]);
      else if (fd.t === 'image') facts.push([fd.l, '(picture)']);
      else if (fd.t === 'ref') facts.push([fd.l, (find(v) || {}).title || v]);

      else if (fd.t === 'checkbox') facts.push([fd.l, 'Yes']);
      else if (fd.t === 'link') facts.push([fd.l, v.l || v.u]);
      else if (fd.t === 'list') facts.push([fd.l, v.length + ' item(s)']);
      else facts.push([fd.l, String(v)]);
    });
    const ver = rec.verification || {};
    el.innerHTML = `<div class="box"><div class="hd"><h2>Preview</h2><button class="btn sm" data-p="close">Close</button></div>

  /* ---------- relationship picker ---------- */
  function pick(types, current, multi, title, selfId) {
    return new Promise((resolve) => {
      const sel = new Set(Array.isArray(current) ? current : current ? [current] : []);
      const el = document.createElement('div');
      el.className = 'sheet l2';
      el.innerHTML = `<div class="box"><div class="hd"><h2>${esc(title)}</h2><button class="btn sm" data-p="cancel">Cancel</button></div>
        <div class="fld"><input type="search" id="pq" placeholder="Search…" aria-label="Search records"></div><div id="pl"></div>
        <div class="foot"><button class="btn p" data-p="done">Done</button>${multi ? '' : '<button class="btn" data-p="clear">Clear</button>'}</div></div>`;
      document.body.appendChild(el);
      const draw = () => {
        const q = norm($('#pq', el).value);
        const items = [].concat(...types.map((t) => st.data[t] || [])).filter((r) => r.id !== selfId && (!q || norm(r.title + ' ' + (r.subtitle || '')).indexOf(q) >= 0))
          .sort((a, b) => (sel.has(b.id) - sel.has(a.id)) || String(a.title).localeCompare(String(b.title), undefined, { numeric: true })).slice(0, 80);
        $('#pl', el).innerHTML = items.map((r) => `<label class="pk"><input type="${multi ? 'checkbox' : 'radio'}" name="pk" value="${esc(r.id)}" ${sel.has(r.id) ? 'checked' : ''}><span>${esc(r.title)}<small>${esc(S.TYPES[r.type].label)}${r.subtitle ? ' · ' + esc(r.subtitle) : ''}</small></span></label>`).join('') || '<p class="empty">Nothing to choose from yet. Create those records first.</p>';
      };
      draw();
      $('#pq', el).oninput = draw;
      el.onchange = (e) => {
        const i = e.target;
        if (i.name !== 'pk') return;
        if (!multi) sel.clear();
        if (i.checked) sel.add(i.value); else sel.delete(i.value);
      };
      el.onclick = (e) => {
        const b = e.target.closest('button');
        if (!b) return;
        const a = b.getAttribute('data-p');
        el.remove();
        if (a === 'cancel') resolve(null);
        else if (a === 'clear') resolve(multi ? [] : '');
        else resolve(multi ? Array.from(sel) : (Array.from(sel)[0] || ''));
      };
    });
  }

'selected' : ''}>View only (default)</option><option value="download" ${ft.download === 'download' ? 'selected' : ''}>Download allowed</option></select><div class="hint">Each file can still be changed individually.</div></div>`;

Every item always has a title, description, keywords, image, verification and expiry; add the extra information it needs.</p>'}</div>

      if (!st.data[owner]) st.data[owner] = [];

      if (!isExt) { st.grp = 'custom'; st.typ = 'features'; }
      render();

      else if (c === 'delete') {
        if ((st.data[ft.id] || []).length) { showErr(['This feature still has ' + st.data[ft.id].length + ' record(s). Delete or archive them first, or set the feature to Hidden instead.']); return; }
        if (!confirm('Delete the feature "' + ft.name + '"?')) return;
        st.cfg.features = st.cfg.features.filter((x) => x.id !== ft.id); delete st.data[ft.id]; S.configure(st.cfg);
        saveWork(); closeOverlay(); st.typ = 'features'; toast('Feature deleted. Publish to apply.'); render();
      } else if (fb) {

        if (fb === 'del') {
          const f = ft.fields[i];

          ft.fields.splice(i, 1);
        } else { const j = fb === 'up' ? i - 1 : i + 1; [ft.fields[i], ft.fields[j]] = [ft.fields[j], ft.fields[i]]; }
        draw();
      }
    };
    overlay.onchange = (e) => {
      if (e.target.classList && e.target.classList.contains('ft')) { read(); sync(); draw(); }
    };
  }

  var FIELD_KEY = /^c_[a-z0-9_]{1,30}$/;
  var FEATURE_ID = /^[a-z][a-z0-9]*(-[a-z0-9]+)*$/;
  // Upload rules shared by the Editor (pre-check) and the Worker (authoritative check).
  var LIMITS = { imageMB: 3, thumbKB: 300, rawImageMB: 15, docMB: 15 };
  var DOC_EXT = { pdf: 'application/pdf', doc: 'application/msword', docx: 'application/vnd.openxmlformats-officedocument.wordprocessingml.document', xls: 'application/vnd.ms-excel', xlsx: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet', ppt: 'application/vnd.ms-powerpoint', pptx: 'application/vnd.openxmlformats-officedocument.presentationml.presentation' };
  var isImageKey = function (v) { return typeof v === 'string' && IMG_KEY.test(v) && v.indexOf('..') < 0; };
  var isDocKey = function (v) { return typeof v === 'string' && DOC_KEY.test(v) && v.indexOf('..') < 0; };
  var isImageRef = function (v) { return isImageKey(v) || (typeof v === 'string' && isUrl(v)); };

  function slugify(s) {
    return String(s || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '')
      .replace(/&/g, ' and ').replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '');
  }
  function slugOf(rec) { return rec.id.slice(rec.type.length + 1); }

  function refFields(type) {
    return TYPES[type].fields.filter(function (x) { return x.t === 'ref' || x.t === 'refs'; });
  }

  function expiryMs(r) {
    if (!r.expires) return null;
    var t = Date.parse(r.expires);
    return isNaN(t) ? null : t;
  }
  // draft | active | expiring | expired | archived
  function lifecycle(r, now) {
    if (r.status === 'archived') return 'archived';
    if (r.status !== 'published') return 'draft';
    var e = expiryMs(r);
    if (e != null) {
      if (e <= now) return 'expired';
      if (e - now <= 3600000) return 'expiring';
    }
    return 'active';
  }
  function isPublic(r, now) {
    var l = lifecycle(r, now);
    return l === 'active' || l === 'expiring';
  }

  function isUrl(u) {
    try { var x = new URL(u); return x.protocol === 'http:' || x.protocol === 'https:'; } catch (e) { return false; }
  }
  function validDate(s) {
    return DATE_RE.test(s) && !isNaN(Date.parse(s + 'T00:00:00Z'));
  }

  // lookup(id) -> type string or undefined (omit to skip relationship checks)
  function validateRecord(r, lookup) {
    var e = [];
    var T = r && TYPES[r.type];
    if (!T) return ['Unknown record type "' + (r && r.type) + '"'];
    if (!r.title || !String(r.title).trim()) e.push('Title is required');
    if (!r.id || !ID_RE.test(r.id)) e.push('ID must use lowercase letters, numbers and hyphens only');
    else if (r.id.indexOf(r.type + '-') !== 0) e.push('ID must start with "' + r.type + '-"');
    else if (r.id.length === r.type.length + 1) e.push('ID is incomplete');
    if (STATUSES.indexOf(r.status) < 0) e.push('Status must be draft, published or archived');
    var v = r.verification || {};
    var vs = v.status || 'unverified';
    if (!VERIFY[vs]) e.push('Unknown verification status');
    if (isVerified(vs) && !String(v.by || '').trim()) e.push('"Verified by" is required for verified records');
    if (v.date && !validDate(v.date)) e.push('Verification date must be YYYY-MM-DD');
    if (r.image && !isImageRef(r.image)) e.push('Image must be an uploaded image (img/…) or a valid http(s) URL');
    if (r.expires != null && r.expires !== '' && isNaN(Date.parse(r.expires))) e.push('Expiration is not a valid date/time');
    ['keywords', 'aliases', 'related'].forEach(function (k) {
      if (r[k] != null && (!Array.isArray(r[k]) || r[k].some(function (x) { return typeof x !== 'string'; }))) e.push(k + ' must be a list of text values');
    });
    (r.related || []).forEach(function (id) {
      if (lookup && !lookup(id)) e.push('Related record "' + id + '" does not exist');
    });
    T.fields.forEach(function (fd) {
      var val = r[fd.k];
      var empty = val == null || val === '' || (Array.isArray(val) && !val.length) || (fd.t === 'checkbox' && val === false && !fd.req);
      if (fd.req && !fd.hidden && r.status === 'published' && (empty || (fd.t === 'checkbox' && val !== true))) { e.push(fd.l.replace(/ \(.*\)$/, '') + ' is required'); return; }
      if (val == null || val === '' || (Array.isArray(val) && !val.length)) return;
      var bad = function (m) { e.push(fd.l.replace(/ \(.*\)$/, '') + ': ' + m); };
      var chkFile = function (o) {
        if (!o || typeof o !== 'object' || !isDocKey(o.k)) { bad('invalid file reference'); return; }
        if (o.n != null && (typeof o.n !== 'string' || o.n.length > 200)) bad('invalid file name');
        if (o.dl != null && [0, 1, true, false].indexOf(o.dl) < 0) bad('invalid download permission');
        if (fd.accept === 'pdf' && !/\.pdf$/i.test(o.k)) bad('only PDF files are allowed here');
      };
      switch (fd.t) {
        case 'email': if (!EMAIL_RE.test(val)) bad('invalid email address'); break;
        case 'phone': if (!/^[+()\-.\s\d]{7,25}$/.test(val) || String(val).replace(/\D/g, '').length < 7) bad('invalid phone number'); break;
        case 'url': if (!isUrl(val)) bad('invalid link (must start with http:// or https://)'); break;
        case 'date': if (!validDate(val)) bad('use the format YYYY-MM-DD'); break;
        case 'time': if (!TIME_RE.test(val)) bad('use the format HH:MM'); break;
        case 'number': if (!isFinite(Number(val))) bad('must be a number'); break;
        case 'select': if ((fd.opts || []).indexOf(val) < 0) bad('invalid option'); break;
        case 'list': if (!Array.isArray(val)) bad('must be a list'); break;
        case 'datetime': if (isNaN(Date.parse(val))) bad('not a valid date/time'); break;
        case 'location': if (typeof val !== 'string' || val.length > 300) bad('invalid location'); break;
        case 'checkbox': if (typeof val !== 'boolean') bad('must be yes or no'); break;
        case 'image': if (!isImageRef(val)) bad('invalid image'); break;
        case 'file': chkFile(val); break;
        case 'files': if (!Array.isArray(val) || val.length > 20) bad('invalid document list'); else val.forEach(chkFile); break;
        case 'link': if (!val || typeof val !== 'object' || !isUrl(val.u) || (val.l != null && typeof val.l !== 'string')) bad('needs a valid web address'); break;
        case 'ref':
          if (typeof val !== 'string') { bad('invalid reference'); break; }
          if (lookup) {
            var t1 = lookup(val);
            if (!t1) bad('record "' + val + '" does not exist');
            else if (fd.to && fd.to.indexOf(t1) < 0) bad('"' + val + '" is a ' + t1 + ', expected ' + fd.to.join('/'));
          }
          break;
        case 'refs':
          if (!Array.isArray(val)) { bad('invalid reference list'); break; }
          val.forEach(function (id) {
            if (!lookup) return;
            var t2 = lookup(id);
            if (!t2) bad('record "' + id + '" does not exist');
            else if (fd.to && fd.to.indexOf(t2) < 0) bad('"' + id + '" is a ' + t2 + ', expected ' + fd.to.join('/'));
          });
          break;
      }
    });
    if (T.fields.some(function (x) { return x.k === 'lat'; })) {
      if (r.lat !== '' && r.lat != null && Math.abs(Number(r.lat)) > 90) e.push('Latitude must be between -90 and 90');
      if (r.lng !== '' && r.lng != null && Math.abs(Number(r.lng)) > 180) e.push('Longitude must be between -180 and 180');
    }
    return e;
  }

  // data: { office:[...], lecturer:[...], ... } -> { errors:[{id,type,msg}], warnings:[...] }
  function validateAll(data, now) {
    var errors = [], warnings = [];
    var seen = {};
    var types = {};
    TYPE_IDS.forEach(function (t) {
      (data[t] || []).forEach(function (r) { if (r && r.id) types[r.id] = r.type; });
    });
    var lookup = function (id) { return types[id]; };
    (data.__orphans || []).forEach(function (r) { errors.push({ id: (r && r.id) || '?', type: (r && r.type) || '?', msg: 'Unknown record type "' + (r && r.type) + '" (its Custom Feature was removed)' }); });
    TYPE_IDS.forEach(function (t) {
      var arr = data[t] || [];
      if (!Array.isArray(arr)) { errors.push({ id: '-', type: t, msg: 'File for ' + t + ' must be a list' }); return; }
      arr.forEach(function (r, i) {
        if (!r || typeof r !== 'object') { errors.push({ id: '#' + i, type: t, msg: 'Record is not an object' }); return; }
        if (r.type !== t) errors.push({ id: r.id, type: t, msg: 'Record type "' + r.type + '" is stored in the ' + t + ' file' });
        if (r.id) {
          if (seen[r.id]) errors.push({ id: r.id, type: t, msg: 'Duplicate ID' });
          seen[r.id] = 1;
        }
        validateRecord(r, lookup).forEach(function (m) { errors.push({ id: r.id || '#' + i, type: t, title: r.title, msg: m }); });
        // Warn when a published record points at something that will not be public.
        if (r.status === 'published') {
          var chk = function (id) {
            var tg = id && findRec(data, id);
            if (tg && !isPublic(tg, now)) warnings.push({ id: r.id, type: t, title: r.title, msg: 'Links to "' + (tg.title || tg.id) + '", which is not public (' + lifecycle(tg, now) + ')' });
          };
          refFields(t).forEach(function (fd) { (fd.t === 'ref' ? [r[fd.k]] : (r[fd.k] || [])).forEach(chk); });
          (r.related || []).forEach(chk);
        }
      });
    });
    return { errors: errors, warnings: warnings };
  }

  // cfg = { features: [...], extend: { office: [fieldDef,...] } }  -> array of message strings
  function validateConfig(cfg) {
    var e = [];
    if (cfg == null) return e;
    if (typeof cfg !== 'object' || Array.isArray(cfg)) return ['Features file must be an object'];
    var feats = cfg.features || [], ext = cfg.extend || {};
    if (!Array.isArray(feats) || feats.length > 100) return ['features must be a list of at most 100'];
    var ids = {};
    var kinds = BUILDER_KINDS.map(function (k) { return k[0]; });
    var allIds = Object.keys(BASE).concat(feats.map(function (x) { return x && x.id; }));
    var chkFields = function (owner, fields) {
      if (!Array.isArray(fields) || fields.length > 40) { e.push(owner + ': fields must be a list of at most 40'); return; }
      var seen = {};
      fields.forEach(function (d, i) {
        var w = owner + ' field ' + (i + 1) + ': ';
        if (!d || typeof d !== 'object') { e.push(w + 'invalid'); return; }
        if (!FIELD_KEY.test(d.k || '')) e.push(w + 'internal key must look like c_amount');
        else if (seen[d.k]) e.push(w + 'duplicate key ' + d.k);
        seen[d.k] = 1;
        if (!d.l || !String(d.l).trim() || String(d.l).length > 80) e.push(w + 'needs a name (max 80 characters)');
        if (kinds.indexOf(d.t) < 0) e.push(w + 'unknown field type "' + d.t + '"');
        if (d.t === 'select' && (!Array.isArray(d.opts) || !d.opts.length || d.opts.some(function (o) { return typeof o !== 'string' || !o.trim(); }))) e.push(w + 'a dropdown needs at least one option');
        if (d.t === 'related' && d.to && (!Array.isArray(d.to) || d.to.some(function (t) { return allIds.indexOf(t) < 0; }))) e.push(w + 'related record type does not exist');
      });
    };
    feats.forEach(function (ft, i) {
      var w = 'Feature ' + (i + 1) + (ft && ft.name ? ' (' + ft.name + ')' : '') + ': ';
      if (!ft || typeof ft !== 'object') { e.push(w + 'invalid'); return; }
      if (!FEATURE_ID.test(ft.id || '') || ft.id.length > 30) e.push(w + 'id must be lowercase letters, numbers and hyphens');
      else if (BASE[ft.id] || RESERVED.indexOf(ft.id) >= 0) e.push(w + 'id "' + ft.id + '" is reserved');
      else if (ids[ft.id]) e.push(w + 'duplicate id');
      ids[ft.id] = 1;
      if (!ft.name || !String(ft.name).trim() || String(ft.name).length > 80) e.push(w + 'needs a name');
      if (['published', 'hidden'].indexOf(ft.status || 'published') < 0) e.push(w + 'status must be published or hidden');
      if (ft.category && CATEGORY_IDS.indexOf(ft.category) < 0) e.push(w + 'unknown public category');
      if (ft.sort != null && !isFinite(Number(ft.sort))) e.push(w + 'sort order must be a number');
      if (ft.icon && !isImageRef(ft.icon)) e.push(w + 'invalid icon');
      chkFields('Feature "' + ft.name + '"', ft.fields || []);
    });
    Object.keys(ext).forEach(function (t) {
      if (!BASE[t]) { e.push('Custom fields: "' + t + '" is not a built-in record type'); return; }
      chkFields('Fields of ' + BASE[t].label, ext[t]);
    });
    return e;
  }

  function findRec(data, id) {
    for (var i = 0; i < TYPE_IDS.length; i++) {
      var a = data[TYPE_IDS[i]] || [];
      for (var j = 0; j < a.length; j++) if (a[j] && a[j].id === id) return a[j];
    }
    return null;
  }

  return {
    TYPES: TYPES, TYPE_IDS: TYPE_IDS, GROUPS: GROUPS, FEATURES: FEATURES, BASE: BASE, configure: configure, validateConfig: validateConfig,
    featureOf: featureOf, BASE_TYPE_IDS: BASE_TYPE_IDS, typeHidden: typeHidden, typeSearchable: typeSearchable, normField: normField, BUILDER_KINDS: BUILDER_KINDS,
    safeKinds: safeKinds, LIMITS: LIMITS, DOC_EXT: DOC_EXT, isImageKey: isImageKey, isDocKey: isDocKey, isImageRef: isImageRef,
    CATEGORY_IDS: CATEGORY_IDS, RESERVED: RESERVED, FIELD_KEY: FIELD_KEY, STATUSES: STATUSES, VERIFY: VERIFY,
    EXPIRY_PRESETS: EXPIRY_PRESETS, groupOf: groupOf, isVerified: isVerified, expiryFromPreset: expiryFromPreset,
    slugify: slugify, slugOf: slugOf, refFields: refFields, lifecycle: lifecycle, isPublic: isPublic,
    expiryMs: expiryMs, isUrl: isUrl, validateRecord: validateRecord, validateAll: validateAll, findRec: findRec,
    ID_RE: ID_RE
  };
});
